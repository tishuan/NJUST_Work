module regfile(
input clk,
input wen,
input [4:0]raddr1,
input [4:0]raddr2,
input [4:0]waddr,
input [31:0]wdata,
output  [31:0]rdata1,
output  [31:0]rdata2,
input [4:0]test_addr,
output [31:0] test_data

    );
    integer i=0;
    reg[31:0]REG_Files[31:0];
    initial
    for(i=0;i<32;i=i+1)
    REG_Files[i]<=0;
    always @(posedge clk)
    begin
    if(wen)
    REG_Files[waddr]<=wdata;
    end
    assign rdata1=REG_Files[raddr1];
    assign rdata2=REG_Files[raddr2];
    assign test_data=REG_Files[test_addr];
    
Endmodule
